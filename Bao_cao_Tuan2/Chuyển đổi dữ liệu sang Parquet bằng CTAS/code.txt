Sử dụng cú pháp CREATE TABLE AS SELECT (CTAS) để tạo bảng sales_parquet, nén bằng Snappy và giữ nguyên cấu trúc phân vùng.

CREATE TABLE sales_parquet
WITH (
    format = 'PARQUET',
    parquet_compression = 'SNAPPY',
    external_location = 's3://daily-datalake-24112051-24110252-2026/processed/sales/',
    partitioned_by = ARRAY['year', 'month', 'day']
) AS 
SELECT 
    order_id,
    order_time,
    product,
    category,
    quantity,
    unit_price,
    customer_city,
    year,
    month,
    day
FROM sales;
